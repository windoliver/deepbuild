// #594 manual e2e on a real Cargo workspace: the 594 suite's fixture repository (a vendored registry, a heavy
// dependency, probe-macro logging each rustc compile with its --emit and out-dir), with this checkout's real
// scripts/preflight.mjs and scripts/perf.mjs. The change adds dep-b and a test file of its own. Two preflights back to
// back; for each, the full preflight output and every compile by side, split into warm-up and measured run. A
// dependency compile with --emit=dep-info,metadata in a measured run is clippy charged to the measured workload.
//   node e2e-fixture.mjs <checkout>
import { loadavg } from 'node:os';
import { join } from 'node:path';
import { spawnSync } from 'node:child_process';
import { readFileSync, writeFileSync } from 'node:fs';

const checkout = process.argv[2];
const { fixture, side, describe, isDependency, verdict, segments } = await import(join(checkout, 'tests/acceptance/594/helpers.mjs'));
const { fingerprint } = await import(join(checkout, 'scripts/benchmark.mjs'));
const cleanups = [];
const t = { after: (fn) => cleanups.push(fn), diagnostic: () => {} };
const machine = () => { const f = fingerprint(); return `${f.model}, power ${f.power}, lowPower ${f.lowPowerMode}, thermal ${f.thermalState}, load ${loadavg().map((n) => n.toFixed(2)).join(' ')}`; };
try {
  const repo = fixture(t);
  repo.change({ addDependency: true, addTest: true });
  console.log(`fixture repo: <sandbox>/repo; machine: ${machine()}`);
  for (const label of ['first', 'second']) {
    const started = Date.now();
    // repo.preflight() as helpers.mjs runs it, with a 60-minute timeout in place of 15, so the run can wait for the
    // machine's timed grant (granted only while the 1-minute load is under the core count) on a busy machine.
    writeFileSync(repo.log, '');
    const result = spawnSync(process.execPath, ['scripts/preflight.mjs'], { cwd: repo.repo, env: repo.env, encoding: 'utf8', timeout: 3_600_000, maxBuffer: 256 * 1024 * 1024 });
    const run = { status: result.status, output: `${result.stdout ?? ''}${result.stderr ?? ''}${result.error ? `\n${result.error.message}` : ''}`, segments: segments(readFileSync(repo.log, 'utf8'), repo.repo) };
    const seconds = ((Date.now() - started) / 1000).toFixed(1);
    console.log(`\n===== ${label} preflight: exit ${run.status}, ${seconds} s wall; machine after: ${machine()} =====`);
    console.log(run.output.replaceAll(repo.box, '<sandbox>'));
    console.log(`----- compiles by side (${label}) -----`);
    for (const name of ['working tree', 'merge base', 'without own tests']) {
      const s = side(run.segments, name);
      const deps = (list) => list.filter(isDependency);
      const clippyCharged = deps(s.measured).filter(({ emit }) => /^--emit=dep-info,metadata$/.test(emit));
      console.log(`${name}:\n  warm-up dependency compiles: ${deps(s.warm).length} (${describe(deps(s.warm))})\n  measured dependency compiles: ${deps(s.measured).length} (${describe(deps(s.measured))})\n  of which clippy checks charged to the measured run: ${clippyCharged.length}\n  all measured compiles: ${describe(s.measured)}`);
    }
    const v = verdict(run.output);
    console.log(`verdict: status ${v?.status}, current.cpuMs ${v?.current?.cpuMs}, base.cpuMs ${v?.base?.cpuMs}, ratio ${v ? (v.current.cpuMs / v.base.cpuMs).toFixed(3) : 'n/a'} (margin 1.5)`);
  }
} finally {
  for (const fn of cleanups.reverse()) await fn();
}
