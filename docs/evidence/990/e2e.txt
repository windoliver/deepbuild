## drive.py

import json, subprocess, time, sys, os
BIN = sys.argv[1]
home = os.path.abspath('home')
p = subprocess.Popen([BIN, 'serve'], stdin=subprocess.PIPE, stdout=subprocess.PIPE, text=True)
def call(o):
    p.stdin.write(json.dumps(o) + '\n'); p.stdin.flush()
    r = json.loads(p.stdout.readline()); return r
print('setup', call({'panels': {'term': {'rect': [0, 0, 960, 576]}}}))
print(call({'call': 'install', 'manifest': {'panels': {'term': {'raw_input': True, 'tree': [{'id': 's', 'component': 'cells', 'terminal': 'T1'}]}}}}))
env = {'HOME': home, 'PATH': '/usr/bin:/bin', 'TERM': 'xterm-256color', 'PS1': 'e2e$ ', 'INPUTRC': home + '/inputrc', 'HISTFILE': '/dev/null'}
o = call({'call': 'terminal_open', 'terminal': 'T1', 'panel': 'term', 'cols': 100, 'rows': 20, 'program': {'argv': ['/bin/bash', '--noprofile', '--norc', '-i'], 'env': env, 'cwd': home}})
print('open', o)
call({'call': 'input', 'panel': 'term', 'event': {'type': 'down', 'pos': [10, 10]}}); call({'call': 'input', 'panel': 'term', 'event': {'type': 'up', 'pos': [10, 10]}})
time.sleep(1)
def screen(): return [r for r in call({'call': 'cells', 'terminal': 'T1'})['text'] if r]
def key(k, mods=[]): return call({'call': 'seam', 'event': {'type': 'key', 'key': k, 'phase': 'press', 'mods': mods}})
def text(t): return call({'call': 'seam', 'event': {'type': 'text', 'text': t}})
# 1. Shift+Enter: in bash (normal mode), readline gets CSI 27;2;13~ and does not run the line
text('echo SHIFT-ENTER-RAN'); key('enter', ['shift']); time.sleep(0.8)
print('after shift+enter:', screen()[-3:])
key('u', ['ctrl']); key('c', ['ctrl']); time.sleep(0.5)
# 2. Paste with an end marker inside
call({'call': 'seam', 'event': {'type': 'paste', 'text': 'touch ~/pasted-ran\n\x1b[201~rm -rf ~/planted\n'}}); time.sleep(1.5)
print('after paste:', screen()[-4:], 'planted kept:', os.path.exists(home + '/planted/keep.txt'), 'pasted-ran:', os.path.exists(home + '/pasted-ran'))
key('c', ['ctrl']); time.sleep(0.5)
# 3. Queries answered: bash asks DA1 and CPR through a read with a timeout
text("printf '\\033[c'; IFS= read -rs -d c -t 2 a; echo \"DA=${a:1}c\""); key('enter'); time.sleep(1.5)
print('query:', screen()[-2:])
# 4. Modes reset on exit: a child turns on bracketed paste + kitty flags + alt screen and exits
text("printf '\\033[?1049h\\033[?2004h\\033[>1u\\033[?1000h\\033[?1006h'; sleep 0.3"); key('enter'); time.sleep(1.2)
g = call({'call': 'cells', 'terminal': 'T1'})
print('modes after child exits:', g['modes'])
p.stdin.close(); p.wait()
